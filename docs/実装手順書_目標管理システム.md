## 概要

### 実装する機能
- 長期目標・短期目標のCRUD
- 習慣と短期目標の必須紐づけ
- 進捗の自動計算・表示
- 目標達成状態の管理

### データ構造
```
長期目標（LongTermGoal）
  ├─ 短期目標1（ShortTermGoal）
  │   ├─ 習慣A（Habit）
  │   └─ 習慣B（Habit）
  └─ 短期目標2（ShortTermGoal）
      └─ 習慣C（Habit）
```

- Next.js 15 (App Router)
- TypeScript
- Zustand (状態管理)
- Supabase (データベース)
- Tailwind CSS
- React Hook Form + Zod

import { createClient } from '@/lib/supabase'
import type { LongTermGoal, ShortTermGoal } from '@/types'

interface GoalState {
  longTermGoals: LongTermGoal[]
  shortTermGoals: ShortTermGoal[]
  loading: boolean
  error: string | null

  // Long Term Goal Actions
  fetchLongTermGoals: () => Promise<void>
  createLongTermGoal: (data: CreateLongTermGoalData) => Promise<LongTermGoal | null>
  updateLongTermGoal: (id: string, updates: Partial<LongTermGoal>) => Promise<boolean>
  deleteLongTermGoal: (id: string) => Promise<boolean>

  // Short Term Goal Actions
  fetchShortTermGoals: (longTermGoalId?: string) => Promise<void>

  // Progress Calculation
  calculateShortTermGoalProgress: (id: string) => Promise<number>
  calculateLongTermGoalProgress: (id: string) => Promise<number>
}

interface CreateLongTermGoalData {
  title: string

}

interface CreateShortTermGoalData {

  measurement_unit: 'count' | 'days' | 'minutes'
  target_value: number
  deadline?: string
}

export const useGoalStore = create<GoalState>((set, get) => ({
  longTermGoals: [],
  shortTermGoals: [],
  loading: false,
  error: null,

  // 以下、各関数の実装...
}))

### 1-2. 長期目標のCRUD実装

**fetchLongTermGoals**:
```typescript
fetchLongTermGoals: async () => {
  const supabase = createClient()
  set({ loading: true, error: null })

  try {
    const { data: { user } } = await supabase.auth.getUser()
    if (!user) throw new Error('認証が必要です')

    const { data, error } = await supabase
      .from('long_term_goals')
      .select(`
        *,
        short_term_goals (
          *,
          habits (*)
        )
      `)
      .eq('user_id', user.id)
      .order('created_at', { ascending: false })

    if (error) throw error

    set({ longTermGoals: data || [], loading: false })
  } catch (error: any) {
    set({ error: error.message, loading: false })
  }
}
```

**createLongTermGoal**:
```typescript
createLongTermGoal: async (data: CreateLongTermGoalData) => {
  const supabase = createClient()
  set({ loading: true, error: null })

  try {
    const { data: { user } } = await supabase.auth.getUser()
    if (!user) throw new Error('認証が必要です')

    // 期限を計算（今日から period ヶ月後）
    const deadline = new Date()
    deadline.setMonth(deadline.getMonth() + data.period)

    const { data: goal, error } = await supabase
      .from('long_term_goals')
      .insert({
        user_id: user.id,
        title: data.title,
        description: data.description,
        period: data.period,
        deadline: deadline.toISOString(),
        is_achieved: false,
      })
      .select()
      .single()

    if (error) throw error

    set(state => ({
      longTermGoals: [goal, ...state.longTermGoals],
      loading: false
    }))

    return goal
  } catch (error: any) {
    set({ error: error.message, loading: false })
    return null
  }
}
```

**updateLongTermGoal**:
```typescript
updateLongTermGoal: async (id: string, updates: Partial<LongTermGoal>) => {
  const supabase = createClient()
  set({ loading: true, error: null })

  try {
    const { error } = await supabase
      .from('long_term_goals')
      .update(updates)
      .eq('id', id)

    if (error) throw error

    set(state => ({
      longTermGoals: state.longTermGoals.map(g =>
        g.id === id ? { ...g, ...updates } : g
      ),
      loading: false
    }))

    return true
  } catch (error: any) {
    set({ error: error.message, loading: false })
    return false
  }
}
```

**deleteLongTermGoal**:
```typescript
deleteLongTermGoal: async (id: string) => {
  const supabase = createClient()
  set({ loading: true, error: null })

  try {
    // CASCADE削除により、紐づく短期目標・習慣も削除される
    const { error } = await supabase
      .from('long_term_goals')
      .delete()
      .eq('id', id)

    if (error) throw error

    set(state => ({
      longTermGoals: state.longTermGoals.filter(g => g.id !== id),
      loading: false
    }))

    return true
  } catch (error: any) {
    set({ error: error.message, loading: false })
    return false
  }
}
```

fetchShortTermGoals: async (longTermGoalId?: string) => {
  const supabase = createClient()
  set({ loading: true, error: null })

  try {
    const { data: { user } } = await supabase.auth.getUser()
    if (!user) throw new Error('認証が必要です')

    let query = supabase
      .from('short_term_goals')
      .select(`
        *,
        long_term_goal:long_term_goals(*),
        habits(*)
      `)
      .eq('user_id', user.id)

    if (longTermGoalId) {
      query = query.eq('long_term_goal_id', longTermGoalId)
    }

    const { data, error } = await query.order('created_at', { ascending: false })

    if (error) throw error

    set({ shortTermGoals: data || [], loading: false })
  } catch (error: any) {
    set({ error: error.message, loading: false })
  }
}

**実装**:
```typescript
'use client'

import { useState } from 'react'
import type { LongTermGoal } from '@/types'
import { Edit2, Trash2, ChevronRight } from 'lucide-react'
import { useRouter } from 'next/navigation'
import { useGoalStore } from '@/stores/goalStore'
import ProgressBar from '@/components/common/ProgressBar'

interface LongTermGoalCardProps {
  goal: LongTermGoal
}

export default function LongTermGoalCard({ goal }: LongTermGoalCardProps) {
  const router = useRouter()
  const { deleteLongTermGoal, calculateLongTermGoalProgress } = useGoalStore()
  const [progress, setProgress] = useState(0)

  useEffect(() => {
    const loadProgress = async () => {
      const p = await calculateLongTermGoalProgress(goal.id)
      setProgress(p)
    }
    loadProgress()
  }, [goal.id, calculateLongTermGoalProgress])

  // 残り日数を計算
  const remainingDays = Math.ceil(
    (new Date(goal.deadline).getTime() - new Date().getTime()) / (1000 * 60 * 60 * 24)
  )

  // 短期目標の達成状況
  const shortTermGoals = goal.short_term_goals || []
  const achievedCount = shortTermGoals.filter(g => g.is_achieved).length
  const totalCount = shortTermGoals.length

  const handleDelete = async () => {
    if (!confirm(`「${goal.title}」を削除しますか？\n紐づく短期目標と習慣もすべて削除されます。`)) {
      return
    }
    await deleteLongTermGoal(goal.id)
  }

  return (
    <div className="bg-white border border-gray-200 rounded-lg p-6 hover:shadow-lg transition-shadow">
      <div className="flex items-start justify-between mb-4">
        <h3 className="text-lg font-semibold text-gray-900 flex-1">
          {goal.title}
        </h3>
        <div className="flex space-x-2">
          <button
            onClick={() => router.push(`/goals/${goal.id}/edit`)}
            className="p-2 text-gray-500 hover:text-primary-600"
          >
            <Edit2 className="w-4 h-4" />
          </button>
          <button
            onClick={handleDelete}
            className="p-2 text-gray-500 hover:text-red-600"
          >
            <Trash2 className="w-4 h-4" />
          </button>
        </div>
      </div>

      {goal.description && (
        <p className="text-sm text-gray-600 mb-4 line-clamp-2">
          {goal.description}
        </p>
      )}

      <ProgressBar value={progress} className="mb-4" />

      <div className="flex items-center justify-between text-sm mb-4">
        <span className="text-gray-600">
          短期目標: {achievedCount}/{totalCount} 達成
        </span>
        <span className={`font-medium ${
          remainingDays > 30 ? 'text-green-600' :
          remainingDays > 7 ? 'text-yellow-600' :
          'text-red-600'
        }`}>
          残り {remainingDays}日
        </span>
      </div>

      <button
        onClick={() => router.push(`/goals/${goal.id}`)}
        className="w-full flex items-center justify-center space-x-2 py-2 text-primary-600 hover:bg-primary-50 rounded-lg transition-colors"
      >
        <span>詳細を見る</span>
        <ChevronRight className="w-4 h-4" />
      </button>
    </div>
  )
}
```

'use client'

import { useForm } from 'react-hook-form'
import { zodResolver } from '@hookform/resolvers/zod'
import { z } from 'zod'
import { useGoalStore } from '@/stores/goalStore'
import { X } from 'lucide-react'

const schema = z.object({
  title: z.string().min(1, 'タイトルを入力してください'),
  description: z.string().optional(),
  period: z.enum(['3', '6', '12']),
})

type FormData = z.infer<typeof schema>

interface CreateLongTermGoalModalProps {
  isOpen: boolean
  onClose: () => void
}

export default function CreateLongTermGoalModal({ isOpen, onClose }: CreateLongTermGoalModalProps) {
  const { createLongTermGoal } = useGoalStore()
  const { register, handleSubmit, formState: { errors }, reset } = useForm<FormData>({
    resolver: zodResolver(schema) as any,
    defaultValues: {
      period: '3'
    }
  })

  const onSubmit = async (data: FormData) => {
    const result = await createLongTermGoal({
      title: data.title,
      description: data.description,
      period: parseInt(data.period) as 3 | 6 | 12,
    })

    if (result) {
      reset()
      onClose()
    }
  }

  if (!isOpen) return null

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center">
      <div className="absolute inset-0 bg-black/50" onClick={onClose} />
      <div className="relative bg-white rounded-lg p-6 w-full max-w-md">
        <div className="flex items-center justify-between mb-4">
          <h2 className="text-xl font-bold">新しい長期目標</h2>
          <button onClick={onClose} className="p-2 hover:bg-gray-100 rounded">
            <X className="w-5 h-5" />
          </button>
        </div>

        <form onSubmit={handleSubmit(onSubmit)} className="space-y-4">
          <div>
            <label className="block text-sm font-medium text-gray-700 mb-1">
              タイトル *
            </label>
            <input
              {...register('title')}
              className="w-full px-3 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-primary-500"
              placeholder="例: TOEIC 800点を取得する"
            />
            {errors.title && (
              <p className="text-sm text-red-600 mt-1">{errors.title.message}</p>
            )}
          </div>

          <div>
            <label className="block text-sm font-medium text-gray-700 mb-1">
              説明文
            </label>
            <textarea
              {...register('description')}
              className="w-full px-3 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-primary-500"
              rows={3}
              placeholder="目標の詳細や達成したい理由など"
            />
          </div>

          <div>
            <label className="block text-sm font-medium text-gray-700 mb-1">
              期限 *
            </label>
            <div className="grid grid-cols-3 gap-2">
              {[
                { value: '3', label: '3ヶ月' },
                { value: '6', label: '6ヶ月' },
                { value: '12', label: '1年' },
              ].map(option => (
                <label
                  key={option.value}
                  className="flex items-center justify-center p-3 border border-gray-300 rounded-lg cursor-pointer hover:bg-gray-50"
                >
                  <input
                    type="radio"
                    value={option.value}
                    {...register('period')}
                    className="mr-2"
                  />
                  {option.label}
                </label>
              ))}
            </div>
          </div>

          <div className="flex space-x-3">
            <button
              type="button"
              onClick={onClose}
              className="flex-1 py-2 border border-gray-300 rounded-lg hover:bg-gray-50"
            >
              キャンセル
            </button>
            <button
              type="submit"
              className="flex-1 py-2 bg-primary-600 text-white rounded-lg hover:bg-primary-700"
            >
              作成
            </button>
          </div>
        </form>
      </div>
    </div>
  )
}

### 5-1. 進捗更新トリガー

// タスク完了後、紐づく短期目標の進捗を更新
if (task.habit?.short_term_goal_id) {
  const goal = await supabase
    .from('short_term_goals')
    .select('*')
    .eq('id', task.habit.short_term_goal_id)
    .single()

  if (goal.data) {
    let newValue = goal.data.current_value

    switch (goal.data.measurement_unit) {
      case 'count':
        newValue += 1
        break
      case 'minutes':
        newValue += actualDuration
        break
      case 'days':
        // 日数の場合は別途計算が必要
        break
    }

    await useGoalStore.getState().updateShortTermGoalProgress(
      task.habit.short_term_goal_id,
      newValue
    )
  }
}

export function getProgressColor(progress: number): string {
  if (progress >= 100) return 'text-blue-600'
  if (progress >= 71) return 'text-green-600'
  if (progress >= 31) return 'text-yellow-600'
  return 'text-red-600'
}

export function getProgressBarColor(progress: number): string {
  if (progress >= 100) return 'bg-blue-600'
  if (progress >= 71) return 'bg-green-600'
  if (progress >= 31) return 'bg-yellow-600'
  return 'bg-red-600'
}

export function formatProgress(
  currentValue: number,
  targetValue: number,
  unit: 'count' | 'days' | 'minutes'
): string {
  const unitLabels = {
    count: '回',
    days: '日',
    minutes: '分'
  }

  return `${currentValue} / ${targetValue} ${unitLabels[unit]}`
}
