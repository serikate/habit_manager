### 0.4 進捗計算用ストアドプロシージャ

```sql
-- MCP Supabase apply_migration で実行
-- migration name: create_progress_calculation_functions

-- 長期目標の進捗を計算
CREATE OR REPLACE FUNCTION calculate_long_term_goal_progress_v2(p_long_term_goal_id UUID)
RETURNS NUMERIC AS $$
DECLARE
  v_period_months INTEGER;
  v_total_progress NUMERIC := 0;
  v_month RECORD;
BEGIN
  -- 長期目標の期間を取得
  SELECT period_months INTO v_period_months
  FROM long_term_goals
  WHERE id = p_long_term_goal_id;

  IF v_period_months IS NULL OR v_period_months = 0 THEN
    RETURN 0;
  END IF;

  -- 月ごとに計算
  FOR v_month IN
    SELECT
      target_month,
      COUNT(*) as goal_count,
      SUM(CASE WHEN is_achieved THEN 1 ELSE 0 END) as achieved_count
    FROM short_term_goals
    WHERE long_term_goal_id = p_long_term_goal_id
    GROUP BY target_month
  LOOP
    -- この月の貢献度 = 達成数 × (100 / 期間月数 / この月の目標数)
    IF v_month.goal_count > 0 THEN
      v_total_progress := v_total_progress +
        (v_month.achieved_count * (100.0 / v_period_months / v_month.goal_count));
    END IF;
  END LOOP;

  RETURN COALESCE(v_total_progress, 0);
END;
$$ LANGUAGE plpgsql;

-- 1年後ビジョンの進捗を計算
CREATE OR REPLACE FUNCTION calculate_vision_progress(p_vision_id UUID)
RETURNS NUMERIC AS $$
DECLARE
  v_total_progress NUMERIC := 0;
  v_goal_count INTEGER := 0;
  v_goal RECORD;
BEGIN
  FOR v_goal IN
    SELECT id FROM long_term_goals WHERE vision_id = p_vision_id
  LOOP
    v_total_progress := v_total_progress + calculate_long_term_goal_progress_v2(v_goal.id);
    v_goal_count := v_goal_count + 1;
  END LOOP;

  IF v_goal_count = 0 THEN
    RETURN 0;
  END IF;

  RETURN v_total_progress / v_goal_count;
END;
$$ LANGUAGE plpgsql;
```

## Phase 2: ビジョンストアの作成

### 2.1 stores/visionStore.ts を作成

```typescript
import { create } from 'zustand'
import { createClient } from '@/lib/supabase'
import type { Vision, CreateVisionData, UpdateVisionData } from '@/types'

interface VisionState {
  // State
  visions: Vision[]
  loading: boolean
  error: string | null

  // CRUD操作
  fetchVisions: () => Promise<void>
  createVision: (data: CreateVisionData) => Promise<Vision | null>
  updateVision: (id: string, data: UpdateVisionData) => Promise<boolean>
  deleteVision: (id: string) => Promise<boolean>

  // 進捗計算
  calculateVisionProgress: (id: string) => Promise<number>
}

export const useVisionStore = create<VisionState>((set, get) => ({
  // Initial State
  visions: [],
  loading: false,
  error: null,

  fetchVisions: async () => {
    const supabase = createClient()
    set({ loading: true, error: null })
    try {
      const { data: { user } } = await supabase.auth.getUser()
      if (!user) throw new Error('ログインが必要です')

      const { data, error } = await supabase
        .from('visions')
        .select(`
          *,
          long_term_goals (
            *,
            short_term_goals (*)
          )
        `)
        .eq('user_id', user.id)
        .order('created_at', { ascending: false })

      if (error) throw error
      set({ visions: data || [], loading: false })
    } catch (error) {
      const message = error instanceof Error ? error.message : 'ビジョンの取得に失敗しました'
      set({ error: message, loading: false })
    }
  },

  createVision: async (data) => {
    const supabase = createClient()
    set({ loading: true, error: null })
    try {
      const { data: { user } } = await supabase.auth.getUser()
      if (!user) throw new Error('ログインが必要です')

      // 期限を計算
      const deadline = new Date()
      deadline.setMonth(deadline.getMonth() + data.period_months)

      const { data: newVision, error } = await supabase
        .from('visions')
        .insert({
          user_id: user.id,
          title: data.title,
          description: data.description,
          period_months: data.period_months,
          deadline: deadline.toISOString()
        })
        .select()
        .single()

      if (error) throw error

      set(state => ({
        visions: [newVision, ...state.visions],
        loading: false
      }))

      return newVision
    } catch (error) {
      const message = error instanceof Error ? error.message : 'ビジョンの作成に失敗しました'
      set({ error: message, loading: false })
      return null
    }
  },

  updateVision: async (id, data) => {
    const supabase = createClient()
    set({ loading: true, error: null })
    try {
      const updateData: Partial<Vision> = {}

      if (data.title !== undefined) updateData.title = data.title
      if (data.description !== undefined) updateData.description = data.description
      if (data.is_achieved !== undefined) updateData.is_achieved = data.is_achieved

      if (data.period_months !== undefined) {
        updateData.period_months = data.period_months
        const deadline = new Date()
        deadline.setMonth(deadline.getMonth() + data.period_months)
        updateData.deadline = deadline.toISOString()
      }

      const { error } = await supabase
        .from('visions')
        .update(updateData)
        .eq('id', id)

      if (error) throw error

      set(state => ({
        visions: state.visions.map(v =>
          v.id === id ? { ...v, ...updateData } : v
        ),
        loading: false
      }))

      return true
    } catch (error) {
      const message = error instanceof Error ? error.message : 'ビジョンの更新に失敗しました'
      set({ error: message, loading: false })
      return false
    }
  },

  deleteVision: async (id) => {
    const supabase = createClient()
    set({ loading: true, error: null })
    try {
      // 1. 紐づく長期目標を取得
      const { data: longTermGoals } = await supabase
        .from('long_term_goals')
        .select('id')
        .eq('vision_id', id)

      if (longTermGoals && longTermGoals.length > 0) {
        const longTermGoalIds = longTermGoals.map(g => g.id)

        // 2. 紐づく短期目標を取得
        const { data: shortTermGoals } = await supabase
          .from('short_term_goals')
          .select('id')
          .in('long_term_goal_id', longTermGoalIds)

        if (shortTermGoals && shortTermGoals.length > 0) {
          const shortTermGoalIds = shortTermGoals.map(g => g.id)

          // 3. 習慣の紐づけを解除
          await supabase
            .from('habits')
            .update({ short_term_goal_id: null })
            .in('short_term_goal_id', shortTermGoalIds)
        }
      }

      // 4. ビジョンを削除（CASCADE で長期・短期も削除）
      const { error } = await supabase
        .from('visions')
        .delete()
        .eq('id', id)

      if (error) throw error

      set(state => ({
        visions: state.visions.filter(v => v.id !== id),
        loading: false
      }))

      return true
    } catch (error) {
      const message = error instanceof Error ? error.message : 'ビジョンの削除に失敗しました'
      set({ error: message, loading: false })
      return false
    }
  },

  calculateVisionProgress: async (id) => {
    const supabase = createClient()
    try {
      const { data, error } = await supabase
        .rpc('calculate_vision_progress', { p_vision_id: id })

      if (error) throw error
      return data || 0
    } catch (error) {
      console.error('ビジョンの進捗計算に失敗:', error)
      return 0
    }
  }
}))
```

// createLongTermGoal の更新
createLongTermGoal: async (data: CreateLongTermGoalData) => {
  const supabase = createClient()
  set({ loading: true, error: null })
  try {
    const { data: { user } } = await supabase.auth.getUser()
    if (!user) throw new Error('ログインが必要です')

    // ビジョンを取得して期間を計算
    const { data: vision, error: visionError } = await supabase
      .from('visions')
      .select('period_months, created_at')
      .eq('id', data.vision_id)
      .single()

    if (visionError) throw visionError

    // 期間を計算（ビジョンの半分）
    const periodMonths = Math.floor(vision.period_months / 2)

    // 期限を計算
    let deadline: Date
    if (data.period_pattern === 'exact_half') {
      deadline = new Date()
      deadline.setMonth(deadline.getMonth() + periodMonths)
    } else {
      // half_month_end: 半分の月の月末
      deadline = new Date()
      deadline.setMonth(deadline.getMonth() + periodMonths + 1)
      deadline.setDate(0) // 前月の最終日

      .from('long_term_goals')
      .insert({
        user_id: user.id,
        vision_id: data.vision_id,
        title: data.title,
        description: data.description,
        period_months: periodMonths,
        period_pattern: data.period_pattern,
        deadline: deadline.toISOString()

      .select()
      .single()

    if (error) throw error

    set(state => ({
      longTermGoals: [newGoal, ...state.longTermGoals],
      loading: false
    }))

  } catch (error) {
    const message = error instanceof Error ? error.message : '長期目標の作成に失敗しました'

## Phase 4: 進捗計算ロジックの実装

### 4.1 utils/goalProgressCalculator.ts の更新

```typescript
import { ShortTermGoal, LongTermGoal, Vision } from '@/types'

/**
 * 短期目標の進捗率を計算
 */
export function calculateShortTermProgress(goal: ShortTermGoal): number {
  if (goal.target_value <= 0) return 0
  return (goal.current_value / goal.target_value) * 100
}

/**
 * 長期目標の進捗率を計算（新方式）
 * 月ごとの短期目標数に応じて重み付け
 */
export function calculateLongTermProgressV2(
  goal: LongTermGoal,
  shortTermGoals: ShortTermGoal[]
): number {
  if (!goal.period_months || goal.period_months === 0) return 0

  // 月ごとにグループ化
  const goalsByMonth = new Map<string, ShortTermGoal[]>()
  for (const stg of shortTermGoals) {
    const month = stg.target_month
    if (!goalsByMonth.has(month)) {
      goalsByMonth.set(month, [])
    }
    goalsByMonth.get(month)!.push(stg)
  }

  let totalProgress = 0

  for (const [, monthGoals] of goalsByMonth) {
    const goalCount = monthGoals.length
    const achievedCount = monthGoals.filter(g => g.is_achieved).length

    // この月の貢献度 = 達成数 × (100 / 期間月数 / この月の目標数)
    const monthContribution = achievedCount * (100 / goal.period_months / goalCount)
    totalProgress += monthContribution
  }

  return totalProgress
}

/**
 * ビジョンの進捗率を計算
 */
export function calculateVisionProgress(
  vision: Vision,
  longTermGoals: LongTermGoal[],
  shortTermGoalsByLtg: Map<string, ShortTermGoal[]>
): number {
  if (longTermGoals.length === 0) return 0

  let totalProgress = 0

  for (const ltg of longTermGoals) {
    const stgs = shortTermGoalsByLtg.get(ltg.id) || []
    totalProgress += calculateLongTermProgressV2(ltg, stgs)
  }

  return totalProgress / longTermGoals.length
}
```

    if (progress >= 71) return 'bg-green-500'
    if (progress >= 31) return 'bg-yellow-500'
    return 'bg-red-500'
  }

  return (
    <div className="bg-white rounded-xl shadow-md overflow-hidden">
      {/* ヘッダー */}
      <div className="p-4 bg-gradient-to-r from-purple-600 to-blue-600 text-white">
        <div className="flex items-center justify-between">
          <div className="flex-1">
            <div className="flex items-center gap-2">
              <span className="text-2xl">🎯</span>
              <h2 className="text-xl font-bold">{vision.title}</h2>
              {vision.is_achieved && (
                <span className="px-2 py-1 bg-white/20 rounded-full text-sm">
                  達成!
                </span>
              )}
            </div>
            <p className="text-white/80 text-sm mt-1">
              期間: {vision.period_months}ヶ月 |
              期限: {format(new Date(vision.deadline), 'yyyy年M月d日', { locale: ja })}
            </p>
          </div>
          <div className="flex items-center gap-2">
            <button
              onClick={() => setExpanded(!expanded)}
              className="p-2 hover:bg-white/20 rounded-lg"
            >
              {expanded ? <ChevronUp /> : <ChevronDown />}
            </button>
          </div>
        </div>

        {/* 進捗バー */}
        <div className="mt-3">
          <div className="flex justify-between text-sm mb-1">
            <span>進捗</span>
            <span>{progress.toFixed(1)}%</span>
          </div>
          <div className="w-full bg-white/30 rounded-full h-2">
            <div
              className={`h-2 rounded-full ${getProgressColor()}`}
              style={{ width: `${Math.min(progress, 100)}%` }}
            />
          </div>
        </div>
      </div>

      {/* 長期目標リスト */}
      {expanded && (
        <div className="p-4">
          {vision.long_term_goals && vision.long_term_goals.length > 0 ? (

              長期目標がありません
            </p>
          )}

          {/* アクションボタン */}
          <div className="flex justify-end gap-2 mt-4 pt-4 border-t">
            <button
              onClick={() => setShowDeleteConfirm(true)}
              className="flex items-center gap-1 px-3 py-1.5 text-red-600 hover:bg-red-50 rounded-lg text-sm"
            >
              <Trash2 className="w-4 h-4" />
              削除
            </button>
          </div>
        </div>
      )}

      {/* 削除確認ダイアログ */}
      {showDeleteConfirm && (
        <div className="fixed inset-0 bg-black/50 flex items-center justify-center z-50">
          <div className="bg-white p-6 rounded-xl max-w-sm mx-4">
            <h3 className="text-lg font-bold mb-2">ビジョンを削除</h3>

const generateMonthOptions = (longTermGoalDeadline: string) => {
  const options: { value: string; label: string }[] = []
  const now = new Date()
  const deadline = new Date(longTermGoalDeadline)

  let current = new Date(now.getFullYear(), now.getMonth(), 1)
  while (current <= deadline) {
    options.push({
      value: current.toISOString().slice(0, 10),
      label: `${current.getFullYear()}年${current.getMonth() + 1}月`
    })
    current.setMonth(current.getMonth() + 1)
  }

  return options
}

INSERT INTO visions (user_id, title, period_months, deadline, description)

UPDATE long_term_goals ltg
SET
  vision_id = (
