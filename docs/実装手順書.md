### 0.3 データベーススキーマ作成
```sql
-- Supabase SQL Editorで実行

-- Enable RLS
ALTER TABLE auth.users ENABLE ROW LEVEL SECURITY;

-- カテゴリテーブル
CREATE TABLE categories (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  name VARCHAR(100) NOT NULL,
  color VARCHAR(7) DEFAULT '#6366f1',
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 長期目標テーブル
CREATE TABLE long_term_goals (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  title VARCHAR(200) NOT NULL,
  period INTEGER NOT NULL, -- 3,6,12 (months)
  deadline DATE NOT NULL,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 短期目標テーブル
CREATE TABLE short_term_goals (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  long_term_goal_id UUID REFERENCES long_term_goals(id) ON DELETE CASCADE,
  title VARCHAR(200) NOT NULL,
  target_count INTEGER DEFAULT 1,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 習慣項目テーブル
CREATE TABLE habits (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  name VARCHAR(100) NOT NULL,
  category_id UUID REFERENCES categories(id) ON DELETE SET NULL,
  default_duration INTEGER NOT NULL DEFAULT 30, -- minutes
  avg_actual_duration INTEGER DEFAULT 30,
  level INTEGER DEFAULT 1,
  experience_points INTEGER DEFAULT 0,
  is_active BOOLEAN DEFAULT true,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 習慣-目標関連テーブル
CREATE TABLE habit_goal_relations (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  habit_id UUID REFERENCES habits(id) ON DELETE CASCADE,
  short_term_goal_id UUID REFERENCES short_term_goals(id) ON DELETE CASCADE,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 日次タスクテーブル
CREATE TABLE daily_tasks (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  habit_id UUID REFERENCES habits(id) ON DELETE CASCADE,
  date DATE NOT NULL,
  estimated_duration INTEGER NOT NULL,
  importance INTEGER NOT NULL CHECK (importance IN (1, 2, 3)),
  deadline TIMESTAMP WITH TIME ZONE,
  is_recurring BOOLEAN DEFAULT true,
  status VARCHAR(20) DEFAULT 'pending' CHECK (status IN ('pending', 'in_progress', 'completed', 'skipped')),
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 実行履歴テーブル
CREATE TABLE task_executions (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  task_id UUID REFERENCES daily_tasks(id) ON DELETE CASCADE,
  actual_duration INTEGER NOT NULL,
  achievement_rate DECIMAL(5,2) DEFAULT 100.00,
  completed_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 作業時間上限設定テーブル
CREATE TABLE work_limits (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  day_of_week INTEGER NOT NULL CHECK (day_of_week BETWEEN 0 AND 6), -- 0=Sunday
  max_minutes INTEGER NOT NULL DEFAULT 480, -- 8 hours
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- ユーザー設定テーブル
CREATE TABLE user_settings (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  title_mode VARCHAR(20) DEFAULT 'business' CHECK (title_mode IN ('business', 'light')),
  notification_enabled BOOLEAN DEFAULT true,
  theme_mode VARCHAR(10) DEFAULT 'light' CHECK (theme_mode IN ('light', 'dark', 'system')),
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 習慣レビューテーブル
CREATE TABLE habit_reviews (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  habit_id UUID REFERENCES habits(id) ON DELETE CASCADE,
  review_date DATE NOT NULL,
  action VARCHAR(20) NOT NULL CHECK (action IN ('continue', 'modify', 'pause')),
  notes TEXT,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- RLS ポリシー設定
CREATE POLICY "Users can only see their own data" ON categories FOR ALL USING (auth.uid() = user_id);
CREATE POLICY "Users can only see their own data" ON long_term_goals FOR ALL USING (auth.uid() = user_id);
CREATE POLICY "Users can only see their own data" ON short_term_goals FOR ALL USING (auth.uid() = user_id);
CREATE POLICY "Users can only see their own data" ON habits FOR ALL USING (auth.uid() = user_id);
CREATE POLICY "Users can only see their own data" ON daily_tasks FOR ALL USING (auth.uid() = user_id);
CREATE POLICY "Users can only see their own data" ON work_limits FOR ALL USING (auth.uid() = user_id);
CREATE POLICY "Users can only see their own data" ON user_settings FOR ALL USING (auth.uid() = user_id);
CREATE POLICY "Users can only see their own data" ON habit_reviews FOR ALL USING (auth.uid() = user_id);

-- インデックス作成
CREATE INDEX idx_daily_tasks_user_date ON daily_tasks(user_id, date);
CREATE INDEX idx_habits_user_active ON habits(user_id, is_active);
CREATE INDEX idx_task_executions_completed ON task_executions(completed_at);
```

```typescript
// utils/priorityCalculator.ts
export function calculatePriority(importance: number, deadline: Date): number {
  const urgency = calculateUrgency(deadline)
  return importance * urgency
}

export function calculateUrgency(deadline: Date): number {
  const daysUntilDeadline = differenceInDays(deadline, new Date())
  if (daysUntilDeadline <= 0) return 3
  if (daysUntilDeadline <= 3) return 2
  return 1
}
```
